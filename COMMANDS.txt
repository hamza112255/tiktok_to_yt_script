ENCODING COMMANDS
-----------------
# Windows PowerShell
$content = Get-Content client_secret.json -Raw
$bytes = [System.Text.Encoding]::UTF8.GetBytes($content)
$encoded = [Convert]::ToBase64String($bytes)
$encoded | Out-File -FilePath client_secret_b64.txt -NoNewline

# Linux/Mac
cat client_secret.json | base64 > client_secret_b64.txt
cat token.json | base64 > token_b64.txt

========================================
QUICK TROUBLESHOOTING
---------------------
# If script fails, check:
1. Railway logs
2. Environment variables
3. YouTube token validity
4. FFmpeg installation
5. Disk space

# If upload fails:
1. Check YouTube quota
2. Verify token.json
3. Check video format
4. Review Railway logs

# If download fails:
1. Check account URLs
2. Verify yt-dlp works
3. Try with VPN
4. Wait and retry

========================================
EMERGENCY COMMANDS
------------------
# Stop Railway service
# Go to Railway dashboard → Service → Settings → Pause

# Delete all temp files
rmdir /s /q temp_downloads    # Windows
rm -rf temp_downloads         # Linux/Mac

# Reset everything
del token.json
del processed.json
rmdir /s /q temp_downloads

# Reinstall dependencies
pip uninstall -y yt-dlp google-api-python-client
pip install -r requirements.txt
